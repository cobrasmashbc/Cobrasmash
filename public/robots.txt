# Instructions for all web crawlers
User-agent: *

# Allow full access, including CSS, JS and data, so search engines can
# render the page the way visitors see it (news, team and testimonials
# are loaded from /data/ by JavaScript).
Allow: /

# Keep member, news and slider photos out of Google Images. Googlebot can
# still load them to render the page; the logo and icons stay indexable.
User-agent: Googlebot-Image
Disallow: /assets/images/players/
Disallow: /assets/images/news/
Disallow: /assets/images/slider/

# Point all crawlers to the sitemap for a list of important pages
Sitemap: https://cobrasmash.org.uk/sitemap.xml
